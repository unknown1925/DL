X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.25, random_state=42, stratify=y
)

tree_model = DecisionTreeClassifier(
    max_depth=4,
    min_samples_leaf=3,
    random_state=42
)

tree_model.fit(X_train, y_train)

y_pred = tree_model.predict(X_test)

print("Accuracy:", round(accuracy_score(y_test, y_pred), 4))

print("\nClassification Report:\n")
print(classification_report(y_test, y_pred, target_names=target_encoder.classes_))

df["Cluster_ID"] = tree_model.apply(X)

# Create cluster summary
cluster_summary = df.groupby("Cluster_ID").agg(
    Customers=("Loan_ID", "count"),
    Avg_Income=("ApplicantIncome", "mean"),
    Avg_Loan=("LoanAmount", "mean"),
    Avg_Credit_History=("Credit_History", "mean")
).round(2)

# Find dominant loan status in each cluster
dominant_status = df.groupby("Cluster_ID")["Loan_Status"].agg(
    lambda x: x.value_counts().idxmax()
)

# Add to summary
cluster_summary["Dominant_Loan_Status"] = dominant_status

# Sort clusters by size
cluster_summary = cluster_summary.sort_values(by="Customers", ascending=False)
